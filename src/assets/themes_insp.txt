    function adjustHex(hex, amount) {
      let r = parseInt(hex.slice(1, 3), 16);
      let g = parseInt(hex.slice(3, 5), 16);
      let b = parseInt(hex.slice(5, 7), 16);
      r = Math.min(255, Math.max(0, r + amount));
      g = Math.min(255, Math.max(0, g + amount));
      b = Math.min(255, Math.max(0, b + amount));
      return '#' + [r, g, b].map(v => Math.round(v).toString(16).padStart(2, '0')).join('');
    }

    function mixHex(hex1, hex2, weight) {
      const r1 = parseInt(hex1.slice(1, 3), 16);
      const g1 = parseInt(hex1.slice(3, 5), 16);
      const b1 = parseInt(hex1.slice(5, 7), 16);
      const r2 = parseInt(hex2.slice(1, 3), 16);
      const g2 = parseInt(hex2.slice(3, 5), 16);
      const b2 = parseInt(hex2.slice(5, 7), 16);
      const r = Math.round(r1 * (1 - weight) + r2 * weight);
      const g = Math.round(g1 * (1 - weight) + g2 * weight);
      const b = Math.round(b1 * (1 - weight) + b2 * weight);
      return '#' + [r, g, b].map(v => v.toString(16).padStart(2, '0')).join('');
    }

    function getRelativeLuminance(hex) {
      const rgb = [
        parseInt(hex.slice(1, 3), 16) / 255,
        parseInt(hex.slice(3, 5), 16) / 255,
        parseInt(hex.slice(5, 7), 16) / 255
      ].map(c => (c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4)));
      return 0.2126 * rgb[0] + 0.7152 * rgb[1] + 0.0722 * rgb[2];
    }

    function getBestContrastText(bgHex) {
      return getRelativeLuminance(bgHex) > 0.42 ? '#090e17' : '#ffffff';
    }

    /* ==========================================================================
       VIEW RENDERING & INTERACTION HANDLERS
       ========================================================================== */
    function renderStage() {
      const stage = document.getElementById('stage');
      const viewGridBtn = document.getElementById('view-grid');
      const viewSingleBtn = document.getElementById('view-single');
      const singleThemeCtrls = document.getElementById('single-theme-controls');
      const singleModeCtrls = document.getElementById('single-mode-controls');

      viewGridBtn.classList.toggle('active', state.view === 'grid');
      viewSingleBtn.classList.toggle('active', state.view === 'single');

      if (state.view === 'single') {
        document.body.classList.add('single-mode-active');
        singleThemeCtrls.classList.remove('hidden-ctrl');
        singleModeCtrls.classList.remove('hidden-ctrl');

        document.querySelectorAll('#single-theme-controls .ctrl-btn').forEach(btn => {
          btn.classList.toggle('active', btn.dataset.theme === state.theme);
        });
        document.querySelectorAll('#single-mode-controls .ctrl-btn').forEach(btn => {
          btn.classList.toggle('active', btn.dataset.mode === state.mode);
        });

        const themeObj = THEMES.find(t => t.id === state.theme);
        const modeObj = MODES.find(m => m.id === state.mode);

        stage.innerHTML = `
          <div class="single-wrapper">
            <div class="single-header-bar">
              <span>${themeObj.name} · ${modeObj.label}</span>
              <button class="focus-trigger" onclick="setView('grid')">← Back to All 18</button>
            </div>
            <div class="phone-frame">
              ${renderScreenHTML(state.theme, state.mode)}
            </div>
          </div>
        `;
      } else {
        document.body.classList.remove('single-mode-active');
        singleThemeCtrls.classList.add('hidden-ctrl');
        singleModeCtrls.classList.add('hidden-ctrl');

        let html = `<div class="matrix-container">`;

        THEMES.forEach(theme => {
          html += `
            <section class="theme-section" id="section-${theme.id}">
              <div class="theme-section-header">
                <div class="theme-section-title">
                  <span>${theme.name}</span>
                  <span class="theme-section-tag">${theme.desc}</span>
                </div>
                <span class="text-xs font-semibold text-slate-400 hidden sm:inline">3 Modes</span>
              </div>
              <div class="modes-row">
          `;

          MODES.forEach(mode => {
            html += `
              <div class="render-card-wrap">
                <div class="render-meta">
                  <span>${theme.name} — ${mode.label}</span>
                  <button class="focus-trigger" onclick="inspectRender('${theme.id}', '${mode.id}')">Focus ↗</button>
                </div>
                <div class="phone-frame">
                  ${renderScreenHTML(theme.id, mode.id)}
                </div>
              </div>
            `;
          });

          html += `
              </div>
            </section>
          `;
        });

        html += `</div>`;
        stage.innerHTML = html;
      }
    }

    function setView(viewName) {
      state.view = viewName;
      renderStage();
      if (viewName === 'single') {
        window.scrollTo({ top: 0, behavior: 'smooth' });
      }
    }

    function selectSingleTheme(themeId) {
      state.theme = themeId;
      renderStage();
    }

    function selectSingleMode(modeId) {
      state.mode = modeId;
      renderStage();
    }

    function inspectRender(themeId, modeId) {
      state.theme = themeId;
      state.mode = modeId;
      state.view = 'single';
      renderStage();
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    function setMonoHue(hex, dotEl) {
      state.monoHex = hex;
      document.querySelectorAll('.swatch-dot').forEach(d => d.classList.remove('active'));
      if (dotEl) dotEl.classList.add('active');
      renderStage();
    }

    document.addEventListener('DOMContentLoaded', () => {
      renderStage();
    });
